```C
vector<int> preorderTraversal(TreeNode* root) {
    stack<TreeNode*> st;
    vector<int> result;
    if (root == NULL) return result;
    st.push(root);
    while (!st.empty()) {
        TreeNode* node = st.top();                       // 中
        st.pop();
        result.push_back(node->val);
        if (node->right) st.push(node->right);           // 右（空节点不入栈）
        if (node->left) st.push(node->left);             // 左（空节点不入栈）
    }
    return result;
}
```

中序遍历：左中右

- [94.二叉树的中序遍历](https://leetcode.cn/problems/binary-tree-inorder-traversal/)
- 递归法

```C
void Traversal(TreeNode* cur, vector<int>& res){
    if(cur == nullptr)
        return ;
    Traversal(cur->left, res);
    res.push_back(cur->val);
    Traversal(cur->right, res);
}
```

- 迭代法：中序遍历的迭代写法与前序不同，它是先访问左子树直接到底部把左子树都入栈，然后慢慢返回处理，返回处理的时候再把右子树入栈，每个节点都是这样

```C
vector<int> inorderTraversal(TreeNode* root) {
    stack<TreeNode*> st;
    vector<int> res;
    TreeNode* cur = root;
    while(!st.empty() || cur){
        if(cur){
            st.push(cur);
            cur = cur->left;
        }
        else{
            cur = st.top();
            st.pop();
            res.push_back(cur->val);
            cur = cur->right;
        }
    }
    return res;
}
```

后序遍历：左右中

- [145.二叉树的后序遍历](https://leetcode.cn/problems/binary-tree-postorder-traversal)
- 563.二叉树的坡度

- 递归法

```C
void Traversal(TreeNode* cur, vector<int>& res){
    if(cur == nullptr)
        return ;
    Traversal(cur->left, res);
    Traversal(cur->right, res);
    res.push_back(cur->val);
}
```

- 迭代法:直接root入栈，处理节点，左右入栈，弹出就是右左。之后翻转一下就是左右中

```C
vector<int> postorderTraversal(TreeNode* root) {
    stack<TreeNode*> st;
    vector<int> result;
    if (root == NULL) return result;
    st.push(root);
    while (!st.empty()) {
        TreeNode* node = st.top();
        st.pop();
        result.push_back(node->val);
        if (node->left) st.push(node->left); // 相对于前序遍历，这更改一下入栈顺序 （空节点不入栈）
        if (node->right) st.push(node->right); // 空节点不入栈
    }
    reverse(result.begin(), result.end()); // 将结果反转之后就是左右中的顺序了
    return result;
}
```

### 统一迭代法实现前中后序遍历

- 以中序为例子，要访问的节点放入栈中，右中左的顺序放入，要处理的节点也放入栈中做一个标记，利用nullptr做标记

```C
vector<int> inorderTraversal(TreeNode* root) {
    stack<TreeNode*> st;
    vector<int> res;
    if(root)
        st.push(root);
    while(!st.empty()){
        TreeNode* tmp = st.top();
        if(tmp){
            st.pop();

            /*
            后续
            st.push(tmp);
            st.push(nullptr);
             */
            if(tmp->right)
                st.push(tmp->right);
            /*
            中序
            st.push(tmp);
            st.push(nullptr);
             */
            if(tmp->left)
                st.push(tmp->left);

            /*
            前序
            st.push(tmp);
            st.push(nullptr);
             */
        }else{
            st.pop();
            tmp = st.top();
            st.pop();
            res.push_back(tmp->val);
        }
    }
    return res;
}
```

### 层序遍历- 广度优先遍历

- 借用一个辅助数据结构即队列来实现，队列先进先出，符合一层一层遍历的逻辑，而用栈先进后出适合模拟深度优先遍历也就是递归的逻辑
- 层序遍历就是图论中的广度优先遍历

- [429.N叉树的层序遍历](https://leetcode.cn/problems/n-ary-tree-level-order-traversal/)

```C
vector<vector<int>> levelOrder(TreeNode* root) {
    queue<TreeNode*> que;
    vector<vector<int>> res;
    if(root)
        que.push(root);
    while(!que.empty()){
        int size = que.size();
        vector<int> vec;
        for(int i = 0; i < size; i++){
            TreeNode* tmp = que.front();
            que.pop();
            vec.push_back(tmp->val);
            if(tmp->left)
                que.push(tmp->left);
            if(tmp->right)
                que.push(tmp->right);
        }
        res.push_back(vec);
    }
    return res;
}
```

- 递归法：

```C
void order(TreeNode* cur, vector<vector<int>>& res, int depth){
    if(!cur)
        return ;
    if(res.size() == depth)
        res.push_back(vector<int>());
    res[depth].push_back(cur->val);
    order(cur->left, res, depth + 1);
    order(cur->right, res, depth + 1);
}
vector<vector<int>> levelOrder(TreeNode* root) {
    vector<vector<int>> res;
    order(root, res, 0);
    return res;
}
```

### 遍历中处理的变化——翻转

- [226. 翻转二叉树](https://leetcode.cn/problems/invert-binary-tree/)
- 在遍历的过程中，处理不再是输出，而是交换左右子树

```C
// 递归
TreeNode* invertTree(TreeNode* root) {
    if (root == NULL) return root;
    swap(root->left, root->right);  // 中
    invertTree(root->left);         // 左
    invertTree(root->right);        // 右
    return root;
}
// 迭代——深度优先遍历，前序为例子
TreeNode* invertTree(TreeNode* root) {
    stack<TreeNode*> st;
    if (root != NULL) st.push(root);
    while (!st.empty()) {
        TreeNode* node = st.top();
        if (node != NULL) {
            st.pop();
            if (node->right) st.push(node->right);  // 右
            if (node->left) st.push(node->left);    // 左
            st.push(node);                          // 中
            st.push(NULL);
        } else {
            st.pop();
            node = st.top();
            st.pop();
            swap(node->left, node->right);          // 节点处理逻辑
        }
    }
    return root;
}
// 广度优先遍历——层序遍历
TreeNode* invertTree(TreeNode* root) {
    queue<TreeNode*> que;
    if (root != NULL) que.push(root);
    while (!que.empty()) {
        int size = que.size();
        for (int i = 0; i < size; i++) {
            TreeNode* node = que.front();
            que.pop();
            swap(node->left, node->right); // 节点处理
            if (node->left) que.push(node->left);
            if (node->right) que.push(node->right);
        }
    }
    return root;
}
```

- 递归改成中序是无法实现的，因为已经交换了，会重复翻转。而迭代是使用栈实现的，而非指针。
