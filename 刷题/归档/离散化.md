- 特指整数的离散化
- 数据的值域比较大，个数比较少【一般l和 r较小的话,比如<=10^5,用前缀和】
- 把值域里的数映射到连续的自然数序列里
- 整个的值域跨度很大，但是值比价稀疏
- 这个映射其实就是把数组的值映射到其对应的下标
- 当我们只关心数据的大小关系时，用排名代替原数据进行处理的一种预处理方法。离散化本质上是一种哈希，它在保持原序列大小关系的前提下把其映射成正整数。当原数据很大或含有负数、小数时，难以表示为数组下标，一些算法和数据结构（如BIT）无法运作，这时我们就可以考虑将其离散化。
- 用来离散化的可以是大整数、浮点数、字符串……等等。

---

- A序列中可能有重复元素，需要去重
- 如何算出A[i]离散化之后的值

```C
vector<int> alls;
sorts(alls.begin(),alls.end());
alls.erase(unique(alls.begin()alls.end()),alls.end());//去除重复元素
//unique 将所有的重复元素去重，返回最后不重复的元素下标，将其至最后的位置删除
//二分求出x对应的离散化的值
int find(int x){
    int l=0,r=alls.size()-1;
    while(l<r){
        int mid=(r-l>>1)+r;
        if(alls[mid]>=x)
            r=mid;
        else
            l=mid+1;
    }
    return r+1;
}
```

假定有一个无限长的数轴，数轴上每个坐标上的数都是0现在，我们首先进行 n次操作，每次操作将某一位置 x上的数加 c接下来，进行 m次询问，每个询问包含两个整数 l和 r，你需要求出在区间 [l,r]之间的所有数的和。

- **数组下标比较小的时候，可以用前缀和**

```C
\#include <iostream>
\#include <vector>
\#include <algorithm>
using namespace std;

//将涉及到的所有坐标存入alls[]
//对alls[]离散完之后,alls[]中的值就是所有用到坐标的排序
//之后进行坐标所在位置值add，ADD的时候，需要在alls[]中找到坐标对应的下标，然后输入对应的Add值
//之后的A就是按照坐标顺序的输入值了
//对A[]求前缀和S[],之后对应的l,r在alls[]里面找到坐标Kl、Kr，S[kl]-s[kr-1]即可

//坐标有x、l、r，最多300000个
const int N = 300010;
int m,n;
int A[N],S[N];
vector<int> alls;
vector<pair<int, int>> add,query;

//找到离散化之后对应的排序坐标,+1。是因为对A[]、S[]都是从1开始的
int find(int x){
    int l=0,r=alls.size()-1;
    while(l<r){
        int mid=(r-l)/2+l;
        if(alls[mid]<x)
            l=mid+1;
        else
            r=mid;
    }
    return r+1;
}
int main(){
    ios::sync_with_stdio(false);

    cin>>n>>m;
    //输入至离散化坐标数组、add数组
    for(int i=0;i<n;i++){
        int x,c;
        cin>>x>>c;
        add.push_back(make_pair(x,c));
        alls.push_back(x);
    }
    //输入值离散化坐标数组、pair数组
    for(int i=0;i<m;i++){
        int l,r;
        cin>>l>>r;
        alls.push_back(l);
        alls.push_back(r);
        query.push_back(make_pair(l,r));
    }

    //对所有坐标离散化
    sort(alls.begin(),alls.end());
    alls.erase(unique(alls.begin(),alls.end()),alls.end());//去除重复元素

    //对应坐标添加c，需要在add里面找到x
    //两种写法
    for(vector<pair<int,int>>::iterator it=add.begin();it!=add.end();it++)
        A[find((*it).first)]+=(*it).second;
    // for(auto it:add)
    //     A[find(it.first)]+=it.second;

    //前缀和
    for(int i=1;i <= alls.size();i++)
        S[i]=A[i]+S[i-1];
    for(auto it:query)
        cout<<S[find(it.second)]-S[find(it.first)-1]<<endl;

    return 0;
}
```
