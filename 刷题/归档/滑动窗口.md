- [76. 最小覆盖子串](https://leetcode.cn/problems/minimum-window-substring/)

```C
class Solution {
public:

    string minWindow(string s, string t) {
        unordered_map<char, int> t_char2num, window_char2num;
        for(auto x : t){
            ++t_char2num[x];
        }
        int valid = 0;
        int left = 0, right = 0;
        int min_str_len = INT_MAX, start = 0;
        while(right < s.size()){
            char c = s[right++];
            if(t_char2num.count(c)){
                window_char2num[c]++;
                if(window_char2num[c] == t_char2num[c])
                    ++valid;
            }
            //判断合法个数是否等于target的map.size()
            while(valid == t_char2num.size()){
                if(right - left < min_str_len){
                    start = left;
                    min_str_len = right - left;
                }
                char d = s[left++];
                if(t_char2num.count(d)){
                    if(window_char2num[d] == t_char2num[d])
                        valid--;
                    window_char2num[d]--;
                }
            }
        }
        return min_str_len == INT_MAX ? "" : s.substr(start, min_str_len);
    }
};
```

- [209. 长度最小的子数组](https://leetcode.cn/problems/minimum-size-subarray-sum)

```C
class Solution {
public:
    int minSubArrayLen(int target, vector<int>& nums) {
        int l = 0, r = 0;
        int Min = 0x3f3f3f3f;
        int sum = 0;
        while(r < nums.size()){
            sum += nums[r++];
            while(sum >= target){
                Min = min(Min, r - l);
                sum -= nums[l++];
            }
        }
        return Min == 0x3f3f3f3f ? 0 : Min;
    }
};
```
