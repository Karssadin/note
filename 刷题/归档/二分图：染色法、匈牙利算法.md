- 785.判断二分图

- 二分图是一种图论模型，它将一个图的节点集合分为两个不相交的子集，称为二分图。二分图中的边通常连接不同子集中的节点。它可以用来表示各种问题，如人员分配、任务分配等
- 图中不存在奇数环
- 集合内部没有边，边全部在外部

### 染色法

- 染色法是一种基于图的算法，可以用于判断一个无向图是否为二分图，即能否将图的所有节点分为两个集合，使得每个节点与同一集合中的节点没有边相连。
- 染色法可以使用深度优先搜索或广度优先搜索实现。具体来说，该算法先选定一个节点为起始节点，并将其染成一种颜色（如红色），然后从该节点开始遍历图的所有节点，并将相邻节点染成另一种颜色（如绿色）。如果在遍历的过程中发现某个节点的相邻节点已经被染成了与该节点同一种颜色，则说明该图不是二分图。
- 无论使用深度优先搜索还是广度优先搜索，染色法的时间复杂度都是O(N+E)，其中N表示节点数，E表示边数。但是，由于染色法需要遍历整个图，因此其空间复杂度也比较高，是O(N)。

---

```C
// 染色法使用STL实现
\#include <bits/stdc++.h>
using namespace std;

const int N=100010,M=200010;

int n,m;
int h[N],e[M],ne[M],idx;
int color[N];
void add(int a,int b){
    e[idx]=b;
    ne[idx]=h[a];
    h[a]=idx++;
}
bool dfs(int node,int c){
    color[node]=c;
    for(int i=h[node];i!=-1;i=ne[i]){
        int j=e[i];
        if(!color[j]){
            if(!dfs(j,c^1))
                return false;
        }
        else if(color[j]==c)
            return false;

    }
    return true;
}

int main(){
    cin>>n>>m;

    memset(h,-1,sizeof(h));
    while(m--){
        int a,b;
        cin>>a>>b;
        add(a,b);
        add(b,a);
    }
    bool flag=true;
    for(int i=1;i<=n;i++){
        if(!color[i]){
            if(!dfs(i,0)){
                flag=false;
                break;
            }
        }
    }
    if(flag)
        puts("Yes");
    else
        puts("No");
    return 0;

}
```

### 匈牙利算法

- 匈牙利算法是一种图论算法，用于解决最大匹配问题，它是基于增广路的方法实现的。
- 增广路算法一般采用深度优先搜索（DFS）实现，而匈牙利算法也是如此。具体来说，匈牙利算法在寻找增广路时，会遍历左侧节点的所有右侧未匹配节点，并递归寻找增广路，直到找到一个未匹配的右侧节点，或者找到一个右侧节点，它的所有连边都已经匹配。找到增广路后，就可以根据增广路调整匹配状态，并继续寻找下一个增广路。
- 因此，匈牙利算法使用深度优先搜索实现，其时间复杂度为O(VE)，其中V表示左侧节点数，E表示边数。

---

```C
\#include <bits/stdc++.h>

using namespace std;

const int N = 510 , M = 100010;
int n1,n2,m;
int h[N],ne[M],e[M],idx;
bool st[N];
int match[N];

void add(int a , int b)
{
    e[idx] = b, ne[idx] = h[a], h[a] = idx++;
}

void init()
{
    memset(h,-1,sizeof h);
}

int find(int x)
{
    //遍历自己喜欢的女孩
    for(int i = h[x] ; i != -1 ;i = ne[i])
    {
        int j = e[i];
        if(!st[j])//如果在这一轮模拟匹配中,这个女孩尚未被预定
        {
            st[j] = true;//那x就预定这个女孩了
            //如果女孩j没有男朋友，或者她原来的男朋友能够预定其它喜欢的女孩。配对成功
            if(!match[j]||find(match[j]))
            {
                match[j] = x;
                return true;
            }

        }
    }
    //自己中意的全部都被预定了。配对失败。
    return false;
}
int main()
{
    init();
    cin>>n1>>n2>>m;
    while(m--)
    {
        int a,b;
        cin>>a>>b;
        add(a,b);
    }


    int res = 0;
    for(int i = 1; i <= n1 ;i ++)
    {
         //因为每次模拟匹配的预定情况都是不一样的所以每轮模拟都要初始化
          memset(st,false,sizeof st);
        if(find(i))
          res++;
    }

    cout<<res<<endl;
}
```
