- 使用unordered_map实现
- unordered_map存储对应的子节点，插入时如果有对应char，就更换tmp节点，继续迭代，直到没有，插入新节点，设置isword
- 搜索就判断是否为end()
- 判断前缀类似搜索

```C++
class TrieNode{
public:
    bool isword;
    unordered_map<char, TrieNode*> children;
    TrieNode():isword(false){}	
};
class Trie {
public:
    TrieNode* root;
    Trie() {
        root = new TrieNode();
    }
    
    void insert(string word) {
        TrieNode* tmp = root;
        for(char ch : word){
            if(tmp->children.find(ch) == tmp->children.end())
                tmp->children[ch] = new TrieNode();
            tmp = tmp->children[ch];
        }
        tmp->isword = true;
    }
    
    bool search(string word) {
        TrieNode* tmp = root;
        for(char ch: word){
            if(tmp->children.find(ch) == tmp->children.end())
                return false;
            tmp = tmp->children[ch];
        }

        return tmp->isword;
    }
    
    bool startsWith(string prefix) {
        TrieNode* tmp = root;
        for(char ch: prefix){
            if(tmp->children.find(ch) == tmp->children.end())
                return false;
            tmp = tmp->children[ch];
        }
        return true;
    }
};
```

```C++
\#include <iostream>
\#include <string>
\#include <unordered_map>
\#include <memory>

using namespace std;

template <typename T>
class TrieNode{
public:
    bool is_value_node{false};
    unordered_map<char, shared_ptr<TrieNode>> children;

    virtual ~TrieNode() = default;
    virtual shared_ptr<TrieNode<T>> clone() const {
        return make_shared<TrieNode<T>> (*this);
    }
};

template <typename T>
class TrieNodeWithValue : public: TrieNode<T>{
public:
    T value;
    bool is_value_node{true};

    TrieNodeWithValue(const T& value): value(value){}

    shared_ptr<TrieNode<T>> clone() const {
        return make_shared<TrieNodeWithValue<T>> (*this);
    }

};

template <typename T>
class Trie{
private:
    shared_ptr<TrieNode<T>> root;
public:
    Trie(){
        root = make_shared<TrieNode<T>>();
    } 
    Trie Puts(string key, T value) const {
        if(key.empty)
            return {};

        shared_ptr<TrieNode<T>> newRoot = root ? root->clone() : nullptr;
        shared_ptr<TrieNode<T>> current = root;

        for(char ch: key){
            auto tmp = current.find(ch);
            if(tmp != current.end())
                current->children[ch] = tmp->second->clone();
            else
                current->children[ch] = make_shared<TrieNode<T>>();
            current = current->children[ch];
        }

        current = make_shared<TrieNodeWithValue<T>>(move(value));
        return Trie(newRoot);
    }   
}
```
