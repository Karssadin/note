template <typename NameType,typename AgeType>
class Person{
public:
	//类内实现
	Person(NameType name,AgeType age){
		this->Name=name;
		this->Age=age;
	}
	NameType Name;
	AgeType Age;
	void show(){
		cout<<this->Name<<this->Age;
	}
};
//构造函数类外实现
template <typename NameType,typename AgeType>
Person<NameType,AgeType> ::	Person(NameType name,AgeType age){
	this->Name=name;
	this->Age=age;

}
//成员函数类外实现
template <typename NameType,typename AgeType>
void Person<NameType,AgeType> ::show(){
	cout<<this->Name<<this->Age;
}
