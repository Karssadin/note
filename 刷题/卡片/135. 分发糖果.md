- 两次遍历，让分数比旁边高的糖果多

```C
int candy(vector<int>& ratings) {
    vector<int> candyV(ratings.size(), 1);
    for(int i = 1; i < ratings.size(); i++){
        if(ratings[i] > ratings[i - 1])
            candyV[i] = candyV[i - 1] + 1;
    }
    for(int i = ratings.size() - 2; i >= 0; i--)
        if(ratings[i] > ratings[i + 1])
            candyV[i] = max(candyV[i], candyV[i + 1] + 1);
    int res = 0;
    for(auto x: candyV)
        res += x;
    return res;
}
```
