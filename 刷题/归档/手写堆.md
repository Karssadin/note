---
- 堆是一个完全二叉树，结点从左到右依次排布
- 小根堆：每个堆都小于等于儿子，根节点是每个堆的最小值
- 存储：使用一维数组存储，1是根节点，x的左儿子是2x，右儿子是2x+1，下标从1开始

---

```C
down(x){
    //寻找和两个儿子相比的最小值，放到树底，直到不可以交换
}
up(x){
    //与根节点比较，如果小于根节点，就进行交换，直到不可以交换
}
```

---

- 插入一个数：heap[++size]=x,up(size);
- 求集合当中的最小值:heap[1];
- 删除最小值:heap[1]=heap[size--]，down(1);
- 删除任意一个元素:heap[i]=heap[size--],down或者up;
- 修改任意一个元素:heap[k]=x,down 或者up

//不带映射的普通堆 up size操作

```C
\#include <iostream>
\#include <algorithm>

using namespace std;

const int N = 1e5+10;
int n,m;
int h[N],s;

void down(int u){
    int t=u;
    if(u*2<=s && h[u*2]<h[t])
        t=u*2;
    if(u*2+1<=s && h[u*2+1]<h[t])
        t=u*2+1;
    if(u!=t){
        swap(h[u],h[t]);
        down(t);
    }
}

void up(int u){
    while(u/2 && h[u/2]>h[u]){
        swap(h[u/2],h[u]);
        u/=2;
    }
}
int main(){
    ios::sync_with_stdio(false);
    cin.tie(0);
    cout.tie(0);
    cin>>n>>m;

    for(int i=1;i<=n;i++)
        cin>>h[i];

    s=n;
    //从最后一个非叶子结点进行down操作即可
    for(int i=n/2;i;i--)
        down(i);

    while(m--){
        cout<<h[1]<<" ";
        h[1]=h[s--];
        down(1);
    }
    return 0;
}
```

//带映射关系的堆

```C
\#include <iostream>
\#include <algorithm>
\#include <string.h>
using namespace std;

const int N = 1e5+10;
int n,m;
int h[N],s;
int ph[N];//存储第K个插入点的下标
int hp[N];//堆里面的某个点，是第几个插入的。在交换的时候需要直到堆中点对应的K序列
//ph[j]=k、hp[k]=j;


void heap_swap(int a,int b){
    swap(ph[hp[a]],ph[hp[b]]);
    swap(hp[a],hp[b]);
    swap(h[a],h[b]);
}
void down(int u){
    int t=u;
    if(u*2<=s && h[u*2]<h[t])
        t=u*2;
    if(u*2+1<=s && h[u*2+1]<h[t])
        t=u*2+1;
    if(u!=t){
        heap_swap(u,t);
        down(t);
    }
}

void up(int u){
    while(u/2 && h[u/2]>h[u]){
        heap_swap(u/2,u);
        u/=2;
    }
}
int main(){
    ios::sync_with_stdio(false);
    cin.tie(0);
    cout.tie(0);
    cin>>n;
    while(n--){
        char op[10];
        int k,x;
        cin>>op;
        if(!strcmp(op,"I")){
            ++s;
            ++m;
            cin>>x;
            ph[m]=s;
            hp[s]=m;
            h[s]=x;
            up(s);
        }
        else if(!strcmp(op,"PM"))
            cout<<h[1]<<endl;
        else if(!strcmp(op,"DM")){
            heap_swap(1,s--);
            down(1);
        }
        else if(!strcmp(op,"D")){
            cin>>k;
            k=ph[k];
            heap_swap(k,s);
            s--;
            down(k),up(k);
        }
        else{

            cin>>k>>x;
            k=ph[k];
            h[k]=x;
            down(k),up(k);

        }
    }

    return 0;
}
```
