```C
\#include <bits/stdc++.h>

using namespace std;

int exgcd(int a,int b,int &x,int &y){
    if(!b){
        x=1,y=0;
        return a;
    }
    //由公式得，x=y`，这里x放到y的位置就可以
    int d = exgcd(b,a%b,y,x);

    //y=x`-a/b*y`=y-a/b*x
    y-=a/b*x;
    return d;
}

int main(){
    ios::sync_with_stdio(false);

    cin.tie(0);
    cout.tie(0);

    int n;
    cin>>n;
    while(n--){
        int a,b,x,y;
        cin>>a>>b;

        exgcd(a,b,x,y);
        cout<<x<<" "<<y<<endl;
    }

}
```

### 线性同余方程

```C
\#include<bits/stdc++.h>
using namespace std;
int n, x, y;
using LL = long long ;
int exgcd(int a, int b, int &x, int &y){
    if (!b){
        x = 1, y = 0;
        return a;
    }
    int d = exgcd(b, a % b, x, y);
    int z = x;
    x = y;
    y = z - a / b * y;
    return d;
}

int main(){
    cin >> n;
    while (n --){
        int a, b, m;
        cin >> a >> b >> m;
        int d = exgcd(a, m, x, y);
        if (b % d) puts("impossible");
        else{
            x = (LL)x * b / d % m;
            cout << x << endl;
        }
    }
    return 0;
}
```
