- _**分治思想**_
- 快排使用数来分治，归并是以中间点来分成左右
- 步骤：
    - 确定分界点，`mid=(l+r)>>1`
    - 递归排序左边和右边
    - 把两个有序的序列，合并成一个有序的序列
    - 双指针：两个指针指向的都是同一个序列的最小值，两个指针再比较一下大小，取最下放入结果序列，++；再次进行比较；直到一个序列到终点，把剩下的另一个都补到序列中

---

- 使用归并排序查找逆序数的数量，在合一的时候进行比较，判断右边如果小于左边，就将结果+1+mid-L，代表坐标有这么多大于右边的，为逆序数

```C++

int N;
vector<int> vec;
vector<int> tmp;

void mergeSort(vector<int>& v, int l, int r)
{
    if(l >= r)
    {
        return ;
    }
    int mid = l + ((r - l) >> 1);
    
    mergeSort(v, l, mid);
    mergeSort(v, mid + 1, r);
    
    int i = l;
    int j = mid + 1;
    int k = 0;
    
    while(i <= mid && j <= r)
    {
        if(v[i] <= v[j])
        {
            tmp[k++] = v[i++];
        }
        else
        {
            tmp[k++] = v[j++];
        }
    }
    while(i <= mid)
    {
        tmp[k++] = v[i++];
    }
    while(j <= r)
    {
        tmp[k++] = v[j++];
    }
    
    for(i = l, j = 0; i <= r;)
    {
        vec[i++] = tmp[j++];
    }
}
```

> 求逆序数

```C++
\#include <iostream>
\#include <vector>

using namespace std;

int N;
long long sum = 0;
vector<int> vec;
vector<int> tmp;

void mergeSort(vector<int>& v, int l, int r)
{
    if(l >= r)
    {
        return ;
    }
    int mid = l + ((r - l) / 2);
    
    mergeSort(v, l, mid);
    mergeSort(v, mid + 1, r);
    
    int i = l;
    int j = mid + 1;
    int k = 0;
    
    while(i <= mid && j <= r)
    {
        if(v[i] <= v[j])
        {
            tmp[k++] = v[i++];
        }
        else
        {
            tmp[k++] = v[j++];
            sum += 1 + mid - i;
        }
    }
    while(i <= mid)
    {
        tmp[k++] = v[i++];
    }
    while(j <= r)
    {
        tmp[k++] = v[j++];
    }
    
    for(i = l, j = 0; i <= r;)
    {
        vec[i++] = tmp[j++];
    }
}

int main()
{
    cin >> N;
    vec.resize(N);
    tmp.resize(N);
    for(int i = 0; i < N; i++)
        cin >> vec[i];
    
    mergeSort(vec, 0, N - 1);
    
    cout << sum;
    return 0;
}
```
