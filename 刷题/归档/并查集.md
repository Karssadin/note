---
- 并查集是一种常用的数据结构，用于维护一些元素之间的集合关系。它可以高效地支持以下操作：
- 询问元素是否在一个集合中
- 合并两个集合
- 并查集的思想：
- 代表元素就是同组公共祖先节点

---

- 初始状态下，每个元素都是一个独立的组，即每个元素的父节点为自身。
- 合并两个组时，将其中一个组的**代表元素**的父节点设置为另一个组的代表元素。
- 查询两个元素是否在同一组中时，查找它们的代表元素是否相同即可。

---

- 判断树根：p[x]=x
- 如何求x的集合编号：while(p[x]!=x) x=p[x];
- 如何合并两个集合：p[x]=y

---

- 优化：
- 在构造函数中，我们初始化每个元素的父节点和秩(rank)，初始时每个元素都是单独的一个集合，父节点为它本身，秩为0。
- 在查找操作中，我们使用**路径压缩**的方式来优化，即不断向上查找代表元素，并将路径上的节点直接连接到根节点，这样可以减少后续的查找时间。
- 在合并操作中，我们使用**按秩合并**的方式来优化，即将秩小的根节点连接到秩大的根节点上，这样可以保证树的深度不会过大，从而保证查找的效率。同时，如果两个根节点的秩相同，则将其中一个根节点的秩加1，并合并到这个节点中

```C
\#include <iostream>
\#include <vector>
using namespace std;

const int N = 100010;
int n,m;

class UnionFind {
public:
    UnionFind(int n) {
        parent.resize(n);
        rank.resize(n);
        size.resize(n);
        for (int i = 0; i < n; ++i) {
            parent[i] = i;
            rank[i] = 0;
            size[i]=1;
        }
    }
    int find(int x) {
        if (parent[x] != x) {
            parent[x] = find(parent[x]);  // 路径压缩
        }
        return parent[x];
    }
    void merge(int x, int y) {
        int px = find(x);
        int py = find(y);
        if (px == py) {
            return;
        }
        if (rank[px] < rank[py]) {
            parent[px] = py;
            size[py]+=size[px];
        } else if (rank[px] > rank[py]) {
            parent[py] = px;
            size[px]+=size[py];
        } else {
            parent[px] = py;
            size[py]+=size[px];
            rank[py]++;
        }
    }
    int getsize(int x){
        return size[find(x)];
    }
private:
    std::vector<int> parent;
    std::vector<int> rank;
    std::vector<int> size;
};
int main(){
    cin>>n>>m;
    UnionFind u=UnionFind(n);

    while (m -- ){
        string op;
        cin>>op;

        if(op=="C"){
            int a,b;

            cin>>a>>b;
            u.merge(u.find(a),u.find(b));
        }
        else if(op=="Q1"){

            int a,b;
            cin>>a>>b;
            if(u.find(a)==u.find(b))
                cout<<"Yes"<<endl;
            else
                cout<<"No"<<endl;
        }
        else{
            int x;
            cin>>x;
            cout<<u.getsize(x)<<endl;
        }
    }
    return 0;
}
```
