### 哈希表——Hash

- 之前的离散化是一个特殊的哈希方式，这个是一般意义上的哈希

---

- 哈希表是一种常用的数据结构，用于快速查找数据。它通过将关键字映射为一个整数索引，将数据存储在数组中，从而实现高效的查找操作。
- 哈希表包括两个主要部分：哈希函数和哈希冲突处理方法。
- 哈希函数是将关键字映射为整数索引的方法。常用的哈希函数包括取余法（取质数，离2的整数次幂尽可能地远，这样冲突地概率小）、乘法散列法、斐波那契哈希法等。哈希函数的设计决定了哈希表的性能。
- 哈希冲突处理方法是解决哈希函数可能会将不同的关键字映射到相同的索引位置的问题。
- 常用的哈希冲突处理方法包括开放寻址法和链表法。

---

- 开放寻址法是指当一个位置已经被占用时，继续查找下一个位置，直到找到一个空位置。

---

```C
\#include <iostream>
\#include <cstring>
using namespace std;

const int N = 200003,null=0x3f3f3f3f;

int h[N];
//返回X所在位置，或者应该存储的位置
//h[t]==标志，就说明不存在
int find(int x){
    int t=(x%N+N)%N;
    while(h[t]!=null&&h[t]!=x){
        ++t;
        if(t==N)
            t=0;
    }
    return t;
}
int main(){
    ios::sync_with_stdio(false);
    cin.tie(0);
    cout.tie(0);
    int n;
    cin>>n;

    //每个字节设置为0x3f，int就是0x3f3f3f3f
    memset(h,0x3f,sizeof(h));
    while (n -- ){
        char op;
        int x;
        cin>>op>>x;
        int k=find(x);
        if(op=='I')
            h[k]=x;
        else{
            if(h[k]==x)
                cout<<"Yes"<<endl;
            else
                cout<<"No"<<endl;
        }
    }
}
```

- 链表法是指在哈希表的每个位置上，维护一个链表，将哈希值相同的元素存储在同一个链表中。

0 1 2 3 4 54 57

```C
\#include <iostream>
\#include <cstring>
using namespace std;

const int N = 100003;

int h[N];//存放存储的值的下标，空指针下标为-1
int e[N],ne[N],idx;//e[]存放的为值，ne存放下一个结点的下标

void insert(int x){
    int k=(x%N+N)%N;

    //按0和1插入的都插入 K的位置来说
    //e[0]=0 ne[0]=-1 h[k]=0
    //e[1]=1 ne[1]=0 h[k]=1
    e[idx]=x;
    ne[idx]=h[k];
    h[k]=idx++;
}

bool find(int x){

    int k=(x%N+N)%N;
    for(int i =h[k];i!=-1;i=ne[i])
        if(e[i]==x)
            return true;

    return false;
}
int main(){
    ios::sync_with_stdio(false);
    cin.tie(0);
    cout.tie(0);
    int n;
    cin>>n;
    memset(h,-1,sizeof(h));

    while (n -- ){
        char op;
        int x;
        cin>>op>>x;
        if(op=='I')
            insert(x);
        else{
            if(find(x))
                cout<<"Yes"<<endl;
            else
                cout<<"No"<<endl;
        }
    }
}
```

### 字符串哈希方式

- 字符串前缀hash法
- 映射公式 (X1×Pn−1+X2×Pn−2+⋯+Xn−1×P1+Xn×P0)modQ
- 任意字符不可以映射成0，否则会出现不同的字符串都映射成0的情况，比如A,AA,AAA皆为0
- 冲突问题：通过巧妙设置P (131 或 13331) , Q (2^64)的值，一般可以理解为不产生冲突。
- 问题是比较不同区间的子串是否相同，就转化为对应的哈希值是否相同。
- 求一个字符串的哈希值就相当于求前缀和，求一个字符串的子串哈希值就相当于求部分和。
- ABCDE 与 ABC 的前三个字符值是一样，只差两位，乘上P的二次方把 ABC 变为 ABC00，再用 ABCDE - ABC00 得到 DE 的哈希值。

---

```C
\#include<iostream>
\#include<cstdio>
\#include<string>
using namespace std;
typedef unsigned long long ULL;
const int N = 1e5+5,P = 131;//131 13331
ULL h[N],p[N];
// h[i]前i个字符的hash值
// 字符串变成一个p进制数字，体现了字符+顺序，需要确保不同的字符串对应不同的数字
// P = 131 或  13331 Q=2^64，在99%的情况下不会出现冲突
// 使用场景： 两个字符串的子串是否相同
ULL query(int l,int r){
    return h[r] - h[l-1]*p[r-l+1];
}
int main(){
    int n,m;
    cin>>n>>m;
    string x;
    cin>>x;

    //字符串从1开始编号，h[1]为前一个字符的哈希值
    p[0] = 1;
    h[0] = 0;
    for(int i=0;i<n;i++){
        p[i+1] = p[i]*P;
        h[i+1] = h[i]*P +x[i];      //前缀和求整个字符串的哈希值
    }

    while(m--){
        int l1,r1,l2,r2;
        cin>>l1>>r1>>l2>>r2;
        if(query(l1,r1) == query(l2,r2)) printf("Yes\n");
        else printf("No\n");

    }
    return 0;
}
```
