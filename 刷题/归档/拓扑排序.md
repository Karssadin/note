- 拓扑排序使用的是广度优先搜索（BFS）算法。拓扑排序的目的是对有向无环图(Directed Acyclic Graph, DAG)中的所有节点进行排序，使得对于任何一个有向边 (u, v)，均有 u 在排序列表中出现在 v 的前面。拓扑排序可以用来检测有向图是否有环，如果有环则无法进行拓扑排序。
- 差分约束问题通常使用拓扑排序解决
- 拓扑排序：
- 每次选择入度为 0 的节点进行遍历，并将其从图中删除，同时更新与该节点相邻的节点的入度值。在遍历完所有的入度为 0 的节点后，如果仍有节点未被访问到，说明图中存在环，无法进行拓扑排序
- 将所有入度为0的点放入队列，队列非空时一直执行，判断当前队头指向的元素是否入度为0（入度-1之后），如果是，就入队
- 可以使用vector来进行，vector有clear函数，不需要使用queue的匿名对象来清空res，但是需要使用erase(q.front);

```C
\#include <iostream>
\#include <vector>
\#include <queue>

class Graph {
public:
    Graph(int n) : adjList(n), indegree(n, 0) {}

    void addEdge(int u, int v) {
        adjList[u].push_back(v);
        indegree[v]++;
    }

    vector<int> topologicalSort() {
        queue<int> q;
        for(int i = 0; i < indegree.size(); i++)
            if(indegree[i] == 0)
                q.push(i);

        int visitedCount = 0;
        vector<int> result;

        while(!q.empty()) {
            int curr = q.front();
            q.pop();
            result.push_back(curr);

            for(auto neighbor : adjList[curr]) {
                indegree[neighbor]--;
                if(indegree[neighbor] == 0)
                    q.push(neighbor);
            }

            visitedCount++;
        }

        if(visitedCount != adjList.size())
            return {};

        return res;
    }

private:
    vector<vector<int>> adjList;
    vector<int> indegree;
};
```

```C++
vector<int> findOrder(int numCourses, vector<vector<int>>& prerequisites) {
        vector<vector<int>> adjList(numCourses);
        vector<int> indegree(numCourses);
        queue<int> q;
        vector<int> res;

        for(auto x: prerequisites){
            adjList[x[1]].push_back(x[0]);
            indegree[x[0]]++;
        }

        for(int i = 0; i < numCourses; ++i)
            if(indegree[i] == 0)
                q.push(i);
        int count = 0;
        while(!q.empty()){
            int curr = q.front();
            q.pop();
            res.push_back(curr);

            for(auto x : adjList[curr]){
                indegree[x]--;
                if(indegree[x] == 0)
                    q.push(x);
            }
            count++;
        }
        if(count != numCourses)
            return {};
        return res;
    }
```
