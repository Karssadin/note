- 建立一个通用类，类中数据类型可以不具体指定，用一个虚拟的类型来代表

```C++
template <typename T>//类

//例子
template <typename NameType,typename AgeType>
class Person{
public:
	Person(NameType name,AgeType age){
		this->Name=name;
		this->Age=age;

	}
	NameType Name;
	AgeType Age;
	void show(){
		cout<<this->Name<<this->Age;
	}
};

Person<string,int> p1("tst",11);

//类模板特例话
template <typename T, int line, int column>
class Matrix;

template <typename T>
class Matrix<T, 3, 3>{};// 只适用于行和列都是3的Matrix

template <>
class Matrix<int, 3, 3>{};
```
