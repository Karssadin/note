```C
\#include <bits/stdc++.h>
using namespace std;
typedef long long LL;

int n;

int qmi(int a,int k,int p){
    int res=1;
    while(k){
        //末尾是1的话，就直接*a即可
        if(k&1)
            res=(LL)res*a%p;
        //删除k的末位
        k>>=1;

        //平方 a
        a=(LL)a*a%p;
    }
    return res;
}
int main(){
    ios::sync_with_stdio(false);
    cin.tie(0);
    cout.tie(0);
    cin>>n;
    while(n--){
        int a,k,p;
        cin>>a>>k>>p;
        cout<<qmi(a,k,p)<<endl;
    }
    return 0;
}
```

### 快速幂求逆元

```C
\#include <bits/stdc++.h>
using namespace std;

typedef long long LL;
int n;

int qmi(int a,int k,int p){
    int res=1;
    while(k){
        if(k&1)
            res=(LL)res*a%p;
        k>>=1;
        a=(LL)a*a%p;
    }
    return res;
}

int main(){
    ios::sync_with_stdio(false);
    cin.tie(0);
    cout.tie(0);
    cin>>n;
    while(n--){
        int a,p;
        cin>>a>>p;


        int res=qmi(a,p-2,p);
        if(a%p)
            cout<<res<<endl;
        else
            cout<<"impossible"<<endl;
    }
    return 0;


}
```
