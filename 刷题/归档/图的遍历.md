N皇后问题：全排列的解法。按行枚举当前行可以放在哪里或者按照格子来进行枚举当前格子能不能放

- 见回溯章节，棋盘问题

```C
\#include <iostream>
using namespace std;

const int N = 15;

int n;
char board[N][N]; // 用于存储当前状态下的棋盘

//或者可以使用col[N]来存储当前列有没有皇后：col[i] true表示当前列有皇后
//这个问题中(0,0)是左下角的坐标
//dg[N]来存储右上斜线有没有皇后，右上斜线斜率是1，同一个斜线相同斜率 b=y-x，但是可能为负数，所以+n udg[n+y-x]
//udg[N]来存储左上斜线有没有皇后，左上斜线斜率是-1，同一个斜线相同斜率b=y+x udg[row+i]

// 检查当前位置是否可以放置皇后
bool check(int row, int col) {
    // 检查同一列是否有皇后
    for (int i = 0; i < row; i++)
        if (board[i][col] == 'Q')
            return false;

    // 检查左上方斜线是否有皇后
    for (int i = row - 1, j = col - 1; i >= 0 && j >= 0; i--, j--)
        if (board[i][j] == 'Q')
            return false;

    // 检查右上方斜线是否有皇后
    for (int i = row - 1, j = col + 1; i >= 0 && j < n; i--, j++)
        if (board[i][j] == 'Q')
            return false;

    return true;
}
//检索row行可以放在哪一列
void dfs(int row) {
    if (row == n) {
        // 输出当前状态下的棋盘
        for (int i = 0; i < n; i++) {
            for (int j = 0; j < n; j++)
                cout << board[i][j];
            cout << endl;
        }
        cout << endl;
        return;
    }

    for (int col = 0; col < n; col++) {
        if (check(row, col)) {  // 检查是否可以放置皇后
            board[row][col] = 'Q';
            dfs(row + 1);
            board[row][col] = '.';  // 恢复当前状态
        }
    }
}
int main() {
    cin >> n;

    // 初始化棋盘
    for (int i = 0; i < n; i++) {
        for (int j = 0; j < n; j++) {
            board[i][j] = '.';
        }
    }

    dfs(0);
    return 0;
}
```

- 全排列的解法，按照格子来进行枚举当前格子能不能放

```C
// 不同搜索顺序 时间复杂度不同  所以搜索顺序很重要！
\#include <iostream>
using namespace std;
const int N = 20;

int n;
char g[N][N];
bool row[N], col[N], dg[N], udg[N]; // 因为是一个个搜索，所以加了row

// s表示已经放上去的皇后个数
void dfs(int x, int y, int s)
{
    // 处理超出边界的情况
    if (y == n) y = 0, x ++ ;

    if (x == n) { // x==n说明已经枚举完n^2个位置了
        if (s == n) { // s==n说明成功放上去了n个皇后
            for (int i = 0; i < n; i ++ ) puts(g[i]);
            puts("");
        }
        return;
    }

    // 分支1：放皇后
    if (!row[x] && !col[y] && !dg[x + y] && !udg[x - y + n]) {
        g[x][y] = 'Q';
        row[x] = col[y] = dg[x + y] = udg[x - y + n] = true;
        dfs(x, y + 1, s + 1);
        row[x] = col[y] = dg[x + y] = udg[x - y + n] = false;
        g[x][y] = '.';
    }

    // 分支2：不放皇后
    dfs(x, y + 1, s);
}
int main() {
    cin >> n;
    for (int i = 0; i < n; i ++ )
        for (int j = 0; j < n; j ++ )
            g[i][j] = '.';

    dfs(0, 0, 0);

    return 0;
}
```

```C
//返回以u为根的子树中节点的个数，包括u节点
int dfs(int u) {
    int res = 0; //存储 删掉某个节点之后，最大的连通子图节点数
    st[u] = true; //标记访问过u节点
    int sum = 1; //存储 以u为根的树 的节点数, 包括u，如图中的4号节点

    //访问u的每个子节点
    for (int i = h[u]; i != -1; i = ne[i]) {
        int j = e[i];
        //因为每个节点的编号都是不一样的，所以 用编号为下标 来标记是否被访问过
        if (!st[j]) {
            int s = dfs(j);  // u节点的单棵子树节点数 如图中的size值
            res = max(res, s); // 记录最大联通子图的节点数
            sum += s; //以j为根的树 的节点数
        }
    }

    //n-sum 如图中的n-size值，不包括根节点4；
    res = max(res, n - sum); // 选择u节点为重心，最大的 连通子图节点数
    ans = min(res, ans); //遍历过的假设重心中，最小的最大联通子图的 节点数
    return sum;
}

    // 树中是不存在环的，对于有n个节点的树，必定是n-1条边

        cin >> a >> b;
        add(a, b), add(b, a); //无向图
    }

    cout << ans << endl;
    return 0;
```

### 广度优先

- 对整个空间进行搜索，一层一层的搜索。
- 比DFS稳定
- 如果权重为1，BFS每次搜索到的点，都是距离自己最短的，最短路，其他情况需要用特定算法去做最短路问题。DP是一种特殊的最短路，没有环存在的最短路是DP
- 使用队列：空间占比是指数级别

走迷宫:记录路径的话，就添加一个新的数组，记录下prev[x][y]=t

```C
\#include <iostream>
\#include <cstring>
using namespace std;

const int N = 101;

typedef pair<int, int> PII;

int n,m;
int graph[N][N];
int dis[N][N];
PII q[N*N];

int bfs(){
    int hh=0,tt=0;
    q[0]=make_pair(0,0);
    memset(dis, -1, sizeof(dis));
    dis[0][0]=0;
    int dx[4] = {-1, 0, 1, 0}, dy[4] = {0, 1, 0, -1};

    while(hh<=tt){
        auto t=q[hh++];
        for(int i=0;i<4;i++){
            int x=t.first+dx[i],y=t.second+dy[i];
            if(x>=0&&x<n&&y>=0&&y<m&&graph[x][y]==0&&dis[x][y]==-1){
                dis[x][y]=dis[t.first][t.second]+1;
                q[++tt]=make_pair(x,y);
            }
        }
    }
    return dis[n-1][m-1];
}

int main(){
    cin>>n>>m;
    for(int i=0;i<n;i++){
        for(int j=0;j<m;j++){
            cin>>graph[i][j];
        }
    }

    cout<<bfs()<<endl;
}
```

使用STL作答：如果要记录路径的话，`vector<vector<pair<int, int>>> parent(n, vector<pair<int, int>>(m, {-1, -1}))`

```C
\#include <iostream>
\#include <vector>
\#include <queue>

using namespace std;

int n,m;
int minSteps(vector<vector<int>>& maze){
    int n = maze.size();
    int m = maze[0].size();
    vector<vector<bool>> visited(n, vector<bool>(m, false));
    vector<vector<int>> steps(n, vector<int>(m, 0));

    queue<pair<int, int>> q;
    q.push({0, 0});
    visited[0][0] = true;

    vector<pair<int, int>> directions = {{0, 1}, {0, -1}, {1, 0}, {-1, 0}};

    while (!q.empty()) {
        auto [x, y] = q.front();
        q.pop();

        if (x == n - 1 && y == m - 1) {
            return steps[x][y];
        }

        for (auto [dx, dy] : directions) {
            int nx = x + dx;
            int ny = y + dy;

            if (nx >= 0 && nx < n && ny >= 0 && ny < m && maze[nx][ny] == 0 && !visited[nx][ny]) {
                visited[nx][ny] = true;
                steps[nx][ny] = steps[x][y] + 1;
                q.push({nx, ny});
            }
        }
    }
    return -1; // 无法到达终点
}
int main() {
    vector<vector<int>> graph;
    cin>>n>>m;
    for(int i=0;i<n;i++){
        vector<int> tmp_v;
        for(int j=0;j<m;j++){
            int tmp_in;
            cin>>tmp_in;
            tmp_v.push_back(tmp_in);
        }
        graph.push_back(tmp_v);
    }

    cout << minSteps(graph) << endl; // Output: 8
    return 0;
}
```
