- 指定传入的类型
- 函数参数模板化
- 整个类模板化

```C
//1指定传入的类型：最常用
void printPerson1(Person<string,int>&p){
	p.show();
}
//2函数参数模板化
template<typename T1,typename T2>
void printPerson2(Person<T1,T2>&p){
	p.show();
}
//3 将整个类模板化
template <typename T>
void printPerson3(T &p){
	p.show();
}
```
