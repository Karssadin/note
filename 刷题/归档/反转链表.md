- 双指针法

```C
ListNode* reverseList(ListNode* head) {
    ListNode* _pre = nullptr;
    //前一个结点
    ListNode* _cur = head;
    //当前节点

    //翻转就是让当前节点next = pre之后一直遍历到nullptr
    ListNode* tmp;

    while(_cur != nullptr){
        tmp = _cur->next;
        _cur->next = _pre;
        _pre = _cur;
        _cur = tmp;
    }
    return _pre;
}
```

- 递归实现

```C
ListNode* reverse(ListNode* pre, ListNode* cur){
    if(cur == NULL)
        return pre;
    ListNode* tmp = cur->next;
    cur->next = pre;
    return reverse(cur, tmp);
}
ListNode* reverseList(ListNode* head) {
    return reverse(NULL, head);
}
```
