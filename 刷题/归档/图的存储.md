```C
\#include <cstring>
\#include <iostream>
\#include <algorithm>
using namespace std;

const int N = 100010;
int h[N],e[N],ne[N],idx;
int n;
void add(int a,int b){
    e[idx]=b;
    ne[idx]=h[a];
    h[a]=idx++;
}

int main(){
    //所有头节点指向负一
    cin >>n;

    memset(h,-1,sizeof(h));
    while (n -- ){
        int a,b;
        cin>>a>>b;
        add(a, b);
    }
}
```
